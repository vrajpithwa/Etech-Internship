// Example usage in any component or file import { errorHandler } from './errorHandler'; // Example 1: Try-catch block try { // Your code that might throw an error throw new Error('Something went wrong!'); } catch (error) { errorHandler.handleError(error as Error); } // Example 2: Global error handler window.onerror = (message, source, lineno, colno, error) => { errorHandler.handleError(error || new Error(message.toString())); return true; }; // Example 3: Promise error handling someAsyncFunction() .catch(error => { errorHandler.handleError(error); }); // Example 4: React error boundary class ErrorBoundary extends React.Component { componentDidCatch(error: Error) { errorHandler.handleError(error); } render() { return this.props.children; } }
